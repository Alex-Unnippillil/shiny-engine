cmake_minimum_required(VERSION 3.24)
project(ShinyOptionalSwapper LANGUAGES CXX)
include(CTest)
if(BUILD_TESTING)
  add_executable(swapper-policy-tests ../../tests/swapper/policy.cpp)
  target_include_directories(swapper-policy-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(swapper-policy-tests PRIVATE cxx_std_20)
  add_test(NAME swapper-policy COMMAND swapper-policy-tests)
  if(MSVC)
    target_compile_options(swapper-policy-tests PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(swapper-policy-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(WIN32)
    add_executable(swapper-launch-fixture ../../tests/swapper/fixture.cpp)
    target_compile_definitions(swapper-launch-fixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(swapper-windows-tests ../../tests/swapper/windows.cpp service.cpp dialog.cpp)
    target_include_directories(swapper-windows-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_features(swapper-windows-tests PRIVATE cxx_std_20)
    target_compile_definitions(swapper-windows-tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(swapper-windows-tests PRIVATE /W4 /WX /permissive- /utf-8)
    target_link_libraries(swapper-windows-tests PRIVATE bcrypt comctl32 comdlg32 shell32 ole32 dwmapi)
    add_test(NAME swapper-windows COMMAND swapper-windows-tests $<TARGET_FILE:swapper-launch-fixture>)
    set_tests_properties(swapper-windows PROPERTIES TIMEOUT 45)
  endif()
endif()
