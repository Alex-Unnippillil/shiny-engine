cmake_minimum_required(VERSION 3.24)
project(ShinyDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Test)
qt_standard_project_setup()
include(GNUInstallDirs)
include(CTest)
add_library(desktop-ui STATIC queue.cpp window.cpp)
target_link_libraries(desktop-ui PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
target_include_directories(desktop-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(desktop-ui PUBLIC SHINY_DESKTOP_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(desktop-ui PRIVATE /W4 /WX /utf-8 /permissive-)
else()
  target_compile_options(desktop-ui PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
qt_add_executable(ShinyDesktop WIN32 MACOSX_BUNDLE main.cpp)
target_link_libraries(ShinyDesktop PRIVATE desktop-ui)
set_target_properties(ShinyDesktop PROPERTIES
  MACOSX_BUNDLE_GUI_IDENTIFIER "com.shinyplayer.desktop"
  MACOSX_BUNDLE_BUNDLE_NAME "Shiny Desktop"
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
if(BUILD_TESTING)
  qt_add_executable(desktop-tests ../../tests/desktop/test_desktop.cpp)
  target_link_libraries(desktop-tests PRIVATE desktop-ui Qt6::Test)
  add_test(NAME desktop-policy-ui COMMAND desktop-tests)
  set_tests_properties(desktop-policy-ui PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS ShinyDesktop BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  install(FILES README.md THIRD_PARTY_NOTICES.md ../../LICENSE DESTINATION ShinyDesktop.app/Contents/Resources/notices)
  install(DIRECTORY "${CMAKE_BINARY_DIR}/notices/" DESTINATION ShinyDesktop.app/Contents/Resources/notices OPTIONAL)
else()
  install(FILES README.md THIRD_PARTY_NOTICES.md ../../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/shiny-desktop)
  install(DIRECTORY "${CMAKE_BINARY_DIR}/notices/" DESTINATION ${CMAKE_INSTALL_DATADIR}/shiny-desktop/notices OPTIONAL)
endif()
if(WIN32 OR APPLE)
  qt_generate_deploy_app_script(TARGET ShinyDesktop OUTPUT_SCRIPT deploy_script NO_TRANSLATIONS)
  install(SCRIPT ${deploy_script})
elseif(UNIX)
  install(FILES ../../installers/desktop/shiny-desktop.desktop DESTINATION share/applications)
  # Distro dependencies remain managed by apt; never overwrite system Qt/VLC.
  set(CPACK_GENERATOR "DEB;TGZ")
  set(CPACK_PACKAGE_NAME "shiny-desktop")
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  set(CPACK_PACKAGE_CONTACT "Alex Unnippillil")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Shiny Desktop local-media player (Qt edition)")
  set(CPACK_PACKAGE_FILE_NAME "ShinyDesktop-${PROJECT_VERSION}-Linux-x64")
  set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
  set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Alex Unnippillil")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins, libqt6multimedia6, libqt6multimediawidgets6, gstreamer1.0-plugins-good, gstreamer1.0-libav")
  set(CPACK_DEBIAN_PACKAGE_SECTION "video")
  include(CPack)
endif()
