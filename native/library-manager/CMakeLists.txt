cmake_minimum_required(VERSION 3.24)
project(ShinyLibraryAudit VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_executable(library-policy-tests ../../tests/library-manager/policy.cpp)
target_include_directories(library-policy-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(library-policy-tests PRIVATE cxx_std_20)
add_test(NAME library-policy COMMAND library-policy-tests)
if(MSVC)
  target_compile_options(library-policy-tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
else()
  target_compile_options(library-policy-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(WIN32)
  add_executable(ShinyLibraryAudit main.cpp audit.cpp)
  target_compile_features(ShinyLibraryAudit PRIVATE cxx_std_20)
  target_compile_definitions(ShinyLibraryAudit PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(ShinyLibraryAudit PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(ShinyLibraryAudit PRIVATE bcrypt version wintrust crypt32)
  set_property(TARGET ShinyLibraryAudit PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_library(library-load-canary SHARED ../../tests/library-manager/canary.cpp)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME library-audit-integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/library-manager/integration.py $<TARGET_FILE:ShinyLibraryAudit> $<TARGET_FILE:library-load-canary>)
  set_tests_properties(library-audit-integration PROPERTIES TIMEOUT 90)
endif()
