cmake_minimum_required(VERSION 3.24)
project(ShinyNativeNR LANGUAGES C CXX)
include(CTest)
set(NR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/opendlss-nr" CACHE PATH "Pinned native OpenDLSS source")
add_executable(nr-contract-tests tests.cpp)
target_compile_features(nr-contract-tests PRIVATE cxx_std_20)
target_include_directories(nr-contract-tests PRIVATE "${NR_SOURCE}/src")
add_test(NAME native-nr-contract COMMAND nr-contract-tests)
add_executable(nr-manifest-tests manifest_tests.cpp)
target_compile_features(nr-manifest-tests PRIVATE cxx_std_20)
target_include_directories(nr-manifest-tests PRIVATE "${NR_SOURCE}/src")
add_test(NAME native-nr-manifest COMMAND nr-manifest-tests)
if(WIN32)
  foreach(path "${NR_SOURCE}/src/nr_graph.cpp" "${VULKAN_HEADERS}/include/vulkan/vulkan.h" "${VOLK_SOURCE}/volk.c")
    if(NOT EXISTS "${path}")
      message(FATAL_ERROR "Missing pinned native dependency: ${path}")
    endif()
  endforeach()
  add_executable(nr-guard-tests guard_tests.cpp)
  target_include_directories(nr-guard-tests PRIVATE "${NR_SOURCE}/src")
  target_compile_features(nr-guard-tests PRIVATE cxx_std_20)
  target_compile_definitions(nr-guard-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_test(NAME native-nr-file-guard COMMAND nr-guard-tests)
  # Keep C loader options separate from the C++ integration's forced standard headers.
  add_library(shiny-volk STATIC "${VOLK_SOURCE}/volk.c")
  target_include_directories(shiny-volk PUBLIC "${VULKAN_HEADERS}/include" "${VOLK_SOURCE}")
  target_compile_definitions(shiny-volk PRIVATE VK_ENABLE_BETA_EXTENSIONS)
  set_property(TARGET shiny-volk PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  # Preserve upstream code except UTF-8 Windows file opening and case-insensitive
  # digest spelling. Guarded files are still hashed again by the upstream loader.
  file(READ "${NR_SOURCE}/src/nr_model.cpp" model_cpp)
  string(REPLACE "std::ifstream file(path," "std::ifstream file(std::filesystem::u8path(path)," model_cpp "${model_cpp}")
  string(REPLACE "if (digest != stage[\"sha256\"].str())" "if (_stricmp(digest.c_str(), stage[\"sha256\"].str().c_str()) != 0)" model_cpp "${model_cpp}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nr_model_utf8.cpp" "#include <filesystem>\n${model_cpp}")
  add_executable(ShinyNrWorker main.cpp "${NR_SOURCE}/src/vk_context.cpp" "${CMAKE_CURRENT_BINARY_DIR}/nr_model_utf8.cpp" "${NR_SOURCE}/src/kernels.cpp" "${NR_SOURCE}/src/nr_graph.cpp" "${NR_SOURCE}/src/reference.cpp")
  target_link_libraries(ShinyNrWorker PRIVATE shiny-volk)
  target_include_directories(ShinyNrWorker PRIVATE "${NR_SOURCE}/src" "${VULKAN_HEADERS}/include" "${VOLK_SOURCE}")
  target_compile_features(ShinyNrWorker PRIVATE cxx_std_20)
  target_compile_definitions(ShinyNrWorker PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN VK_ENABLE_BETA_EXTENSIONS _CRT_SECURE_NO_WARNINGS)
  target_compile_options(ShinyNrWorker PRIVATE /EHsc /W3 /utf-8 /FIcstring /FIcstdint)
  set_property(TARGET ShinyNrWorker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
