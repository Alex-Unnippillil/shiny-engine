cmake_minimum_required(VERSION 3.24)
project(ShinyVlcPlayer VERSION 0.10.0 LANGUAGES CXX)
include(CTest)
add_executable(player-core-tests ../../tests/vlc-player/core.cpp)
target_include_directories(player-core-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(player-core-tests PRIVATE cxx_std_20)
add_test(NAME player-core COMMAND player-core-tests)
add_executable(player-layout-tests ../../tests/vlc-player/layout.cpp)
target_include_directories(player-layout-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(player-layout-tests PRIVATE cxx_std_20)
add_test(NAME player-layout COMMAND player-layout-tests)
add_executable(player-preview-tests ../../tests/vlc-player/preview.cpp)
target_include_directories(player-preview-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(player-preview-tests PRIVATE cxx_std_20)
add_test(NAME player-preview COMMAND player-preview-tests)
add_executable(player-workspace-tests ../../tests/vlc-player/workspace.cpp)
target_include_directories(player-workspace-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(player-workspace-tests PRIVATE cxx_std_20)
add_test(NAME player-workspace COMMAND player-workspace-tests)
if(WIN32)
  enable_language(RC)
  add_subdirectory(../library-manager library-manager)
  if(NOT EXISTS "${VLC_INCLUDE_DIR}/vlc/vlc.h")
    message(FATAL_ERROR "Set VLC_INCLUDE_DIR to the official pinned VLC 3.0.24 source include directory.")
  endif()
  add_executable(ShinyVlcPlayer WIN32 main.cpp window.cpp workspace.cpp quick_actions.cpp playback.cpp actions.cpp dialogs.cpp engine.cpp nr_client.cpp nr_panel.cpp managed_preview.cpp selftest.cpp player.rc)
  target_include_directories(ShinyVlcPlayer PRIVATE "${VLC_INCLUDE_DIR}")
  target_compile_features(ShinyVlcPlayer PRIVATE cxx_std_20)
  target_compile_definitions(ShinyVlcPlayer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(ShinyVlcPlayer PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/EHsc;/permissive-;/utf-8>)
  target_link_options(ShinyVlcPlayer PRIVATE /MANIFEST:NO)
  target_link_libraries(ShinyVlcPlayer PRIVATE shiny-package-client comctl32 comdlg32 shell32 ole32 dwmapi d2d1 gdiplus shlwapi uuid uxtheme)
  set_property(TARGET ShinyVlcPlayer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(VLC_TEST_RUNTIME AND VLC_TEST_FIXTURE)
    add_test(NAME vlc-real-playback COMMAND ShinyVlcPlayer --self-test "${VLC_TEST_RUNTIME}" "${VLC_TEST_FIXTURE}" "${CMAKE_BINARY_DIR}/playback-report.json")
    set_tests_properties(vlc-real-playback PROPERTIES TIMEOUT 60)
    add_test(NAME vlc-managed-reference COMMAND ShinyVlcPlayer --managed-test "${VLC_TEST_RUNTIME}" "${VLC_TEST_FIXTURE}" "${CMAKE_BINARY_DIR}/managed-playback-report.json")
    set_tests_properties(vlc-managed-reference PROPERTIES TIMEOUT 60)
  endif()
endif()
