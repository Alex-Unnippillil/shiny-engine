cmake_minimum_required(VERSION 3.24)
project(ShinyEngine LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native companion requires Windows 11 and the Windows SDK.")
endif()
add_executable(shiny-native main.cpp)
target_compile_features(shiny-native PRIVATE cxx_std_20)
target_compile_definitions(shiny-native PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(shiny-native PRIVATE /W4 /EHsc /permissive- /utf-8)
target_link_libraries(shiny-native PRIVATE windowsapp d3d11 dxgi d3dcompiler shcore bcrypt)
set_property(TARGET shiny-native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_test(NAME native-contract COMMAND shiny-native --self-test)
